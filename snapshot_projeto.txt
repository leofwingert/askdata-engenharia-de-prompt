if not api_key:
    raise ValueError("GEMINI_API_KEY não encontrada no arquivo .env!")

client = genai.Client(api_key=api_key)

# Modelo gratuito de embeddings do Google AI Studio
EMBEDDING_MODEL = "gemini-embedding-001"

def extrair_texto_pdf(caminho_pdf: str) -> list[dict]:
    """Le um arquivo PDF e extrai o texto pagina a pagina com metadados."""
    reader = PdfReader(caminho_pdf)
    paginas = []
    nome_arquivo = Path(caminho_pdf).name
    
    for idx, pagina in enumerate(reader.pages):
        texto = pagina.extract_text() or ""
        if texto.strip():
            paginas.append({
                "texto": texto.strip(),
                "arquivo": nome_arquivo,
                "pagina": idx + 1
            })
    return paginas

def extrair_texto_markdown(caminho_md: str) -> list[dict]:
    """Le um arquivo Markdown e extrai o texto com metadados."""
    nome_arquivo = Path(caminho_md).name
    with open(caminho_md, "r", encoding="utf-8") as f:
        texto = f.read().strip()
    
    if texto:
        return [{
            "texto": texto,
            "arquivo": nome_arquivo,
            "pagina": 1
        }]
    return []

def criar_chunks(documentos_paginas: list[dict], chunk_size: int = 1000, chunk_overlap: int = 200) -> list[dict]:
    """Divide os textos em blocos, priorizando o fim de frases."""
    chunks = []

    for item in documentos_paginas:
        texto = item["texto"]
        frases = [parte.strip() for parte in re.split(r"(?<=[.!?])\s+", texto) if parte.strip()]
        inicio_frase = 0
        chunk_idx = 1

        while inicio_frase < len(frases):
            fim_frase = inicio_frase
            tamanho = 0
            while fim_frase < len(frases):
                tamanho_adicional = len(frases[fim_frase]) + (1 if fim_frase > inicio_frase else 0)
                if fim_frase > inicio_frase and tamanho + tamanho_adicional > chunk_size:
                    break
                tamanho += tamanho_adicional
                fim_frase += 1

            trecho = " ".join(frases[inicio_frase:fim_frase])
            
            chunk_id = f"{item['arquivo']}_p{item['pagina']}_c{chunk_idx}"
            chunks.append({
                "id": chunk_id,
                "texto": trecho,
                "arquivo": item["arquivo"],
                "pagina": item["pagina"],
                "chunk_idx": chunk_idx
            })

            proxima_frase = fim_frase
            tamanho_overlap = 0
            while proxima_frase > inicio_frase and tamanho_overlap < chunk_overlap:
                proxima_frase -= 1
                tamanho_overlap += len(frases[proxima_frase]) + 1

            inicio_frase = proxima_frase if proxima_frase > inicio_frase else fim_frase
            chunk_idx += 1

    return chunks

def indexar_no_chromadb(chunks: list[dict], path_db: str = "./chroma_db", collection_name: str = "askdata_knowledge"):
    """Gera embeddings e salva os chunks no ChromaDB com controle de fluxo para evitar erro 429."""
    chroma_client = chromadb.PersistentClient(path=path_db)
    collection = chroma_client.get_or_create_collection(
        name=collection_name,
        metadata={"hnsw:space": "cosine"}
    )
    
    # 1. Recupera os IDs ja persistidos no banco local
    dados_banco = collection.get()
    ids_existentes = set(dados_banco["ids"]) if dados_banco and "ids" in dados_banco else set()
    
    print(f"Total de chunks a processar: {len(chunks)}")
    print(f"Chunks ja salvos no ChromaDB: {len(ids_existentes)}")
    
    for i, ch in enumerate(chunks, 1):
        # Pula itens ja indexados na execucao anterior para poupar cota
        if ch["id"] in ids_existentes:
            continue
            
        tentativas = 3
        while tentativas > 0:
            try:
                res = client.models.embed_content(
                    model=EMBEDDING_MODEL,
                    contents=ch["texto"]
                )
                vetor = res.embeddings[0].values
                
                collection.upsert(
                    ids=[ch["id"]],
                    embeddings=[vetor],
                    documents=[ch["texto"]],
                    metadatas={
                        "arquivo": ch["arquivo"],
                        "pagina": ch["pagina"],
                        "chunk_idx": ch["chunk_idx"]
                    }
                )
                
                # Pausa de 0.7s por chunk: mantem ~85 req/min (abaixo do teto gratuito de 100/min)
                time.sleep(0.2)
                break
                
            except Exception as e:
                if "429" in str(e) or "RESOURCE_EXHAUSTED" in str(e):
                    print(f"\n[Aviso Cota] Teto atingido no chunk {i}. Aguardando 35 segundos para liberar...")
                    time.sleep(35)
                    tentativas -= 1
                else:
                    print(f"\n[Erro fatal no chunk {ch['id']}]: {e}")
                    raise e

        if i % 5 == 0 or i == len(chunks):
            print(f"  -> Progresso: {i}/{len(chunks)} chunks analisados (Total salvo: {collection.count()})...")
            
    print(f"\nIngestao concluida com sucesso! Total definitivo no ChromaDB: {collection.count()} chunks.")

if __name__ == "__main__":
    pasta_dados = "./data"
    todos_documentos = []
    
    # 1. Carregar PDFs da pasta data
    for pdf_path in glob.glob(f"{pasta_dados}/*.pdf"):
        print(f"Processando PDF: {pdf_path}")
        todos_documentos.extend(extrair_texto_pdf(pdf_path))
        
    # 2. Carregar Markdowns da pasta data
    for md_path in glob.glob(f"{pasta_dados}/*.md"):
        print(f"Processando Markdown: {md_path}")
        todos_documentos.extend(extrair_texto_markdown(md_path))
        
    if not todos_documentos:
        print("Nenhum arquivo PDF ou Markdown encontrado em ./data! Adicione arquivos na pasta para testar.")
    else:
        # 3. Gerar Chunks
        lista_chunks = criar_chunks(todos_documentos, chunk_size=1000, chunk_overlap=200)
        # 4. Indexar no ChromaDB
        indexar_no_chromadb(lista_chunks)

import chromadb
from collections import defaultdict

# 1. Conectar ao ChromaDB persistente
client = chromadb.PersistentClient(path="./chroma_db")
collection = client.get_or_create_collection(
    name="askdata_knowledge",
    metadata={"hnsw:space": "cosine"}
)

# 2. Recuperar todos os documentos e metadados indexados
dados = collection.get(include=["documents", "metadatas"])
ids = dados["ids"]
docs = dados["documents"]
metadatas = dados["metadatas"]

total_chunks = len(ids)

print("=" * 60)
print("RELATORIO DE AUDITORIA: CACA AO CHUNK PERDIDO")
print("=" * 60)
print(f"Total de chunks indexados: {total_chunks}")

if total_chunks == 0:
    print("Colecao vazia! Execute 'python src/ingestion.py' primeiro.")
    exit()

# 3. Calcular tamanho medio dos chunks
tamanhos = [len(doc) for doc in docs]
tamanho_medio = sum(tamanhos) / total_chunks
print(f"Tamanho medio dos chunks: {tamanho_medio:.1f} caracteres")
print(f"Menor chunk: {min(tamanhos)} caracteres | Maior chunk: {max(tamanhos)} caracteres")

# 4. Contar chunks por arquivo e por pagina
chunks_por_arquivo = defaultdict(int)
chunks_por_pagina = defaultdict(lambda: defaultdict(int))

for meta in metadatas:
    arquivo = meta.get("arquivo", "desconhecido")
    pagina = meta.get("pagina", 0)
    chunks_por_arquivo[arquivo] += 1
    chunks_por_pagina[arquivo][pagina] += 1

print("\n--- Distribuicao por Arquivo ---")
for arquivo, qtd in chunks_por_arquivo.items():
    print(f"* {arquivo}: {qtd} chunks")
    for pag, count in sorted(chunks_por_pagina[arquivo].items()):
        print(f"    - Pagina {pag}: {count} chunks")

# 5. Identificar chunks cortados no meio da frase (sem pontuacao final)
pontuacao_final = ('.', '!', '?', ':', '"', "'", '”', '’')
chunks_cortados = []

for cid, doc, meta in zip(ids, docs, metadatas):
    texto_limpo = doc.strip()
    if texto_limpo and not texto_limpo.endswith(pontuacao_final):
        chunks_cortados.append((cid, texto_limpo, meta))

print("\n--- Analise de Cortes de Frase ---")
print(f"Chunks sem pontuacao final: {len(chunks_cortados)} de {total_chunks} ({len(chunks_cortados)/total_chunks*100:.1f}%)")

if chunks_cortados:
    print("\nTodos os chunks cortados no meio da frase:")
    for numero, (cid, texto, meta) in enumerate(chunks_cortados, 1):
        fim_trecho = texto[-60:].replace("\n", " ")
        print(f"  {numero}. [ID: {cid}] Arquivo: {meta.get('arquivo')}, Pag: {meta.get('pagina')}")
        print(f"    Final do texto: \"...{fim_trecho}\"")
        print("    " + "-" * 50)

print("\nAuditoria concluida. O overlap configurado na ingestao e de 200 caracteres")
print("aproximados, priorizando a preservacao de frases completas.")

import os
from typing import List, Dict, Any
import chromadb
from dotenv import load_dotenv
from google import genai
from google.genai import types

load_dotenv()
api_key = os.getenv("GEMINI_API_KEY")

if not api_key:
    raise ValueError("GEMINI_API_KEY não encontrada no arquivo .env!")

EMBEDDING_MODEL = "gemini-embedding-001"
MODELO_GERACAO = "gemini-3.5-flash-lite"

class RAGEngine:
    def __init__(self, path_db: str = "./chroma_db", collection_name: str = "askdata_knowledge"):
        """Inicializa a conexão local persistente com o ChromaDB e o cliente Gemini."""
        self.client = genai.Client(api_key=api_key)
        self.chroma_client = chromadb.PersistentClient(path=path_db)
        self.collection = self.chroma_client.get_or_create_collection(
            name=collection_name,
            metadata={"hnsw:space": "cosine"}
        )

    def _gerar_embedding(self, texto: str) -> List[float]:
        """Gera o embedding numérico da pergunta do usuário."""
        res = self.client.models.embed_content(
            model=EMBEDDING_MODEL,
            contents=texto
        )
        return res.embeddings[0].values

    def recuperar_contexto(self, query: str, top_k: int = 4) -> List[Dict[str, Any]]:
        """Recupera os top_k trechos mais próximos via similaridade de cosseno."""
        vetor_query = self._gerar_embedding(query)
        
        resultados = self.collection.query(
            query_embeddings=[vetor_query],
            n_results=top_k
        )
        
        chunks_recuperados = []
        if resultados and resultados.get("documents") and resultados["documents"][0]:
            for doc, meta, dist in zip(
                resultados["documents"][0],
                resultados["metadatas"][0],
                resultados["distances"][0]
            ):
                chunks_recuperados.append({
                    "texto": doc,
                    "arquivo": meta.get("arquivo", "documento_desconhecido"),
                    "pagina": meta.get("pagina", 1),
                    "distancia": dist,
                    "similaridade": round(1.0 - dist, 4)
                })
        return chunks_recuperados

    def responder_pergunta(self, query: str, top_k: int = 4) -> Dict[str, Any]:
        """Executa o pipeline RAG: Recuperação -> Montagem de Contexto -> Síntese Blindada."""
        chunks = self.recuperar_contexto(query, top_k=top_k)
        
        if not chunks:
            return {
                "resposta": "Desculpe, não encontrei informações sobre isso nos documentos fornecidos.",
                "fontes": []
            }

        contexto_formatado = ""
        for idx, ch in enumerate(chunks, 1):
            contexto_formatado += f"\n--- [FONTE {idx} | Arquivo: {ch['arquivo']} | Página: {ch['pagina']}] ---\n"
            contexto_formatado += ch["texto"] + "\n"

        system_instruction = """
Você é o 'AskData', assistente de base de conhecimento da DataLakers.
Sua missão é responder à pergunta do usuário EXCLUSIVAMENTE com base nos trechos fornecidos em <contexto_recuperado>.

REGRAS OBRIGATÓRIAS DE OPERAÇÃO:
1. Responda apenas com fatos descritos no <contexto_recuperado>. Se a pergunta comparar conceitos e eles estiverem descritos nos trechos, você pode sintetizar e contrastar com base nas definições fornecidas.
2. Se o tema principal da pergunta NÃO constar nos trechos fornecidos, você DEVE responder exatamente a seguinte frase:
"Desculpe, não encontrei informações sobre isso nos documentos fornecidos."
3. NUNCA tente inventar, deduzir sem respaldo ou utilizar conhecimentos externos de pré-treino.
4. Para cada ponto abordado, cite explicitamente o nome do arquivo e a página de onde o fato foi extraído (ex: "[Manual_TCESP.pdf, p. 6]").
5. Responda de forma direta, técnica, profissional e em bom português.
"""

        prompt_final = f"""
<contexto_recuperado>
{contexto_formatado}
</contexto_recuperado>

<pergunta_do_usuario>
{query}
</pergunta_do_usuario>
"""

        response = self.client.models.generate_content(
            model=MODELO_GERACAO,
            contents=prompt_final,
            config=types.GenerateContentConfig(
                system_instruction=system_instruction,
                temperature=0.1
            )
        )

        return {
            "resposta": response.text.strip(),
            "fontes": chunks
        }
